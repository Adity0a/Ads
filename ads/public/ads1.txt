Prac2b: XML to horus 

import pandas as pd
import xml.etree.ElementTree as ET

def xml2df(xml_data):
    root = ET.fromstring(xml_data)
    records = []
    for child in root:
        record = {}
        for x in child:
            record[x.tag] = x.text
        records.append(record)
    return pd.DataFrame(records)

InputFile = "C:/Users/ADITYA/Downloads/Ads_Practical/Practical2/prac2b.xml"

Prac4b1: Designing a Routing Diagram for Vermeulen Company

import pandas as pd
from math import radians, cos, sin, asin, sqrt
# Function to calculate distance
def haversine(lon1, lat1, lon2, lat2):
    lon1, lat1, lon2, lat2 = map(radians, [lon1, lat1, lon2, lat2])
    dlon = lon2 - lon1
    dlat = lat2 - lat1
    a = sin(dlat / 2) ** 2 + cos(lat1) * cos(lat2) * sin(dlon / 2) ** 2
    c = 2 * asin(sqrt(a))
    return round(6371 * c, 3) # Distance in km
# Read the CSV file
file_path = "C:/Users/ADITYA/Downloads/Ads_Practical/Practical4/Prac4b_1.csv"
data = pd.read_csv(
file_path,
usecols=["Country", "Place.Name", "Latitude", "Longitude"],
encoding="latin-1"
)
# Remove duplicate records
data = data.drop_duplicates()
# Change column name
data.rename(columns={"Place.Name": "Place_Name"}, inplace=True)
# Create a common column for cross join
data["K"] = 1
# Create all possible combinations of places
cross_data = pd.merge(data, data, on="K")
# Remove K column
cross_data.drop("K", axis=1, inplace=True)
# Rename columns
cross_data.rename(columns={
"Country_x": "Country_from",
"Country_y": "Country_to",
"Place_Name_x": "Place_Name_from",
"Place_Name_y": "Place_Name_to",
"Latitude_x": "Latitude_from",
"Latitude_y": "Latitude_to",
"Longitude_x": "Longitude_from",
"Longitude_y": "Longitude_to"
}, inplace=True)
# Calculate distance
cross_data["Distance_KM"] = cross_data.apply(
lambda row: haversine(
row["Longitude_from"],
row["Latitude_from"],
row["Longitude_to"],
row["Latitude_to"]
),
axis=1
)
# Convert kilometers to miles
cross_data["Distance_Miles"] = cross_data["Distance_KM"] * 0.621371
# Display dimensions
print("Rows and Columns:", cross_data.shape)
# Save the result
output_file ="C:/Users/ADITYA/Downloads/Ads_Practical/Practical4/Prac4b_1_retrieve.csv"
cross_data.to_csv(output_file, index=False, encoding="latin-1")
print("Done!")

Prac4d2: Adopt new shipping containers

import os
import pandas as pd
# Folder location
base = "C:/Users/ADITYA/Downloads/Ads_Practical/Practical4/"
company = 'Prac4d_2'
folder = base + '/' + company + '/01-Retrieve'
# Create folder if it does not exist

    os.makedirs(folder)

# 1. CREATE CONTAINER DATA
containers = []
count = 0
for length in range(1, 21):
    for width in range(1, 10):
        for height in range(1, 6):
            count = count + 1
            volume = length * width * height
            containers.append([
                'Container',
                'C' + format(count, '06d'),
                length,
                width,
                height,
                volume
            ])
ContainerFrame = pd.DataFrame(containers, columns=[
    'ShipType',
    'UnitNumber',
    'Length',
    'Width',
    'Height',
    'ContainerVolume'
])
print('Container Rows:', len(ContainerFrame))
print('Container Columns:', len(ContainerFrame.columns))

    folder + '/Retrieve_Container.csv',
    index=False
)

# 2. CREATE BOX DATA
boxes = []
count = 0
for length in range(1, 21):
    for width in range(1, 21):
        for height in range(1, 21):
            for thickness in range(0, 6):
                box_volume = (length / 10) * (width / 10) * (height / 10)
                product_volume = (
                    (length - thickness) / 10 *
                    (width - thickness) / 10 *
                    (height - thickness) / 10
                )
                if product_volume > 0:
                    count = count + 1
                    boxes.append([
                        'Box',
                        'B' + format(count, '06d'),
                        length / 10,
                        width / 10,
                        height / 10,
                        thickness / 5,
                        round(box_volume, 9),
                        round(product_volume, 9) 
                    ])
BoxFrame = pd.DataFrame(boxes, columns=[
    'ShipType',
    'UnitNumber',
    'Length',
    'Width',
    'Height',
    'Thickness',
    'BoxVolume',
    'ProductVolume'
])
print('Box Rows:', len(BoxFrame))
print('Box Columns:', len(BoxFrame.columns))
BoxFrame.to_csv(
    folder + '/Retrieve_Box.csv',
    index=False
)

# 3. CREATE PRODUCT DATA

count = 0
for length in range(1, 21):
    for width in range(1, 21):
        for height in range(1, 21):
            volume = (
                (length / 10) *
                (width / 10) *
                (height / 10)
            )
            count = count + 1
            products.append([
                'Product',
                'P' + format(count, '06d'),

                width / 10,
                height / 10,

Prac4d3: Creating a delivery route

import os
import pandas as pd
from geopy.distance import geodesic

# File details
input_file = 'Prac4d_3.csv'
output_file = 'Prac4d_3_retrieve.csv'

# Updated folder paths matching your computer
base = "C:/Users/ADITYA/Downloads/Ads_Practical/Practical4/"
company = 'Prac4d_3'
# Input and output paths
input_path = base + '/' + company + '/00-RawData/' + input_file
output_folder = base + '/' + company + '/01-Retrieve'

# Create output folder (Indentation Fixed)
if not os.path.exists(output_folder):
    os.makedirs(output_folder)

# Read warehouse data
warehouse = pd.read_csv(input_path)

# Remove warehouses with latitude or longitude = 0
warehouse = warehouse[
    (warehouse['latitude'] != 0) &
    (warehouse['longitude'] != 0)
]

# Remove duplicate postcodes
warehouse = warehouse.drop_duplicates(subset='postcode')

# Sort by postcode
warehouse = warehouse.sort_values(by='postcode')

# Save cleaned warehouse data
warehouse.to_csv(output_folder + '/' + output_file, index=False)
print('Warehouse data saved!')

# Use first 20 warehouses as sellers
sellers = warehouse.head(20)

# Use first 10,000 warehouses as buyers
buyers = warehouse.head(10000)

# Create routes for each seller
for i in range(len(sellers)):
    seller = sellers.iloc[i]
    print('Run:', i, '======>', seller['postcode'])
    
    # Create a copy of buyer data
    routes = buyers.copy()
    
    # Seller information
    routes['Transaction'] = 'WH-to-WH'
    routes['Seller'] = 'WH-' + str(seller['postcode'])
    routes['Seller_Latitude'] = seller['latitude']
    routes['Seller_Longitude'] = seller['longitude']
    
    # Buyer information
    routes['Buyer'] = 'WH-' + routes['postcode'].astype(str)
    routes['Buyer_Latitude'] = routes['latitude']  # Fixed missing bracket here
    routes['Buyer_Longitude'] = routes['longitude']
    
    # Calculate distance using geodesic miles
    routes['Distance'] = routes.apply(
        lambda row: round(
            geodesic(
                (seller['latitude'], seller['longitude']),
                (row['latitude'], row['longitude'])
            ).miles,
            6
        ),
        axis=1
    )
    
    # Drop raw incoming columns to keep output tidy
    routes = routes.drop(columns=['id', 'postcode', 'latitude', 'longitude'])
    
    output_name = 'Retrieve_Route_WH-' + str(seller['postcode']) + '_Route.csv'
    routes.to_csv(output_folder + '/' + output_name, index=False)

print('Done!!')

Prac6: Processing Data-Builds the time hub, link and satellite

import pandas as pd
import sqlite3 as sq
from datetime import datetime, timedelta
from pytz import timezone
# Connect to database
conn = sq.connect('datavault.db')
# ---------- TIME HUB ----------
dates = [datetime(2018, 1, 1) - timedelta(hours=i) for i in range(10)]

Hub = pd.DataFrame({
'DateTimeKey': [d.strftime('%Y-%m-%d-%H-%M-%S') for d in dates],
'DateTime': [d.strftime('%Y-%m-%d %H:%M:%S') for d in dates]
})
Hub.to_sql('Hub-Time', conn, if_exists='replace', index=False)
print("TIME HUB")
print(Hub)
# ---------- LINK & SATELLITE ----------
for zone in ['Asia/Kolkata', 'Europe/London', 'America/New_York']:
    Satellite = Hub.copy()
    Satellite['LocalTime'] = [
        d.replace(tzinfo=timezone('UTC')).astimezone(timezone(zone)).strftime('%Y-%m-%d%H:%M:%S')
for d in dates
]
Satellite[['DateTimeKey']].to_sql(
'Link-' + zone.replace('/', '-'),
conn, if_exists='replace',
)
Satellite.to_sql(
'Satellite-' + zone.replace('/', '-'),
conn, if_exists='replace', index=False
)
print("\n", zone)
print(Satellite)
conn.close()
print("\nDONE!")
