Prac: resize image to a comman size and generate a video from them

import os
import cv2
from PIL import Image

PATH = r"C:\Users\A\Downloads\ImageProcessing\ImageProcessing\Video"
VIDEO_NAME = "mygeneratedvideo.avi"
VALID_EXTENSIONS = (".jpg", ".jpeg", ".png")


def resize_images_to_mean(folder_path):
    """Calculates the average dimensions of valid images in a folder

    and resizes all of them to that mean size.
    """
    os.chdir(folder_path)

    all_files = os.listdir(".")
    valid_images = [f for f in all_files if f.lower().endswith(VALID_EXTENSIONS)]

    if not valid_images:
        print("No valid images found for resizing.")
        return 0, 0

    total_width = 0
    total_height = 0
    successful_images = 0

    for file in valid_images:
        try:
            with Image.open(file) as im:
                width, height = im.size
                total_width += width
                total_height += height
                successful_images += 1
        except Exception as e:
            print(f"Skipping corrupt file {file}: {e}")
            continue

    if successful_images == 0:
        return 0, 0

    mean_width = int(total_width / successful_images)
    mean_height = int(total_height / successful_images)

    print(f"Mean dimensions calculated: {mean_width}x{mean_height}")

    for file in valid_images:
        try:
            with Image.open(file) as im:
                im_resized = im.resize((mean_width, mean_height))
                im_resized.save(file, "JPEG", quality=95)
                print(f"{file} is resized.")
        except Exception as e:
            print(f"Failed to resize {file}: {e}")

    return mean_width, mean_height


def generate_video(image_folder, video_name, width, height):
    """Generates an AVI video from the resized images."""
    os.chdir(image_folder)

    images = [
        img for img in os.listdir(image_folder) if img.lower().endswith(VALID_EXTENSIONS)
    ]
    print("Images found for video:", images)

    if not images:
        print("No images available to generate video.")
        return

    fourcc = cv2.VideoWriter_fourcc(*"DIVX")
    video = cv2.VideoWriter(video_name, fourcc, 1, (width, height))

    for image in images:
        img_path = os.path.join(image_folder, image)
        frame = cv2.imread(img_path)
        if frame is not None:
            video.write(frame)
        else:
            print(f"Could not read frame for {image}")

    video.release()
    cv2.destroyAllWindows()
    print("Video generation complete!")
