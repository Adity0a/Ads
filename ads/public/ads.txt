Prac2: To implement TCP socket programming using a client-server based program using TCP to find if the entered number is prime

Server.java:

import java.io.*; 
import java.net.*; 
 
class tcpServerPrime { 
    public static void main(String args[]) { 
        try { 
            ServerSocket ss = new ServerSocket(8001); 
            System.out.println("Aaditya's server has started....."); 
            Socket s = ss.accept(); 
            DataInputStream in = new DataInputStream(s.getInputStream()); 
            int num = in.readInt(); 
            DataOutputStream out = new DataOutputStream(s.getOutputStream()); 
             
            boolean flag = false; 
            for (int i = 2; i <= num / 2; ++i) { 
                if (num % i == 0) { 
                    flag = true; 
                    break; 
                } 
            } 
            if (!flag) 
                out.writeUTF(num + " is a prime number."); 
            else 
                out.writeUTF(num + " is not a prime number."); 
             
            s.close(); 
            ss.close(); 
        } catch (Exception e) { 
            System.out.println(e.toString()); 
        } 
    } 
 }

import java.io.*;

class tcpServerOddEven { 
public static void main(String args[]) {
    try {
        ServerSocket ss = new ServerSocket(8001);

        DataInputStream in = new DataInputStream(s.getInputStream());
        int num = in.readInt();

Server.py:

import socket 
host = socket.gethostbyname(socket.gethostname()) 
server = socket.socket(socket.AF_INET, socket.SOCK_DGRAM) 
server.bind((host, 1200)) 
while True: 
    data, address = server.recvfrom(1024)
    msg = data.decode().split()
    op, a, b = msg[0], int(msg[1]), int(msg[2])
    if op == "add":
        res = a + b
    elif op == "sub": 
        res = a - b
    elif op == "div":
        res = a / b
    elif op == "mul":
        res = a * b
    server.sendto(str(res).encode(), (host, 1300))

Prac6b: To implement an RPC-based program to calculate the square, square root, cube, and cube root of a given number.  

Server.py:

import socket 
import math 
 
server = socket.socket(socket.AF_INET, socket.SOCK_DGRAM) 
server.bind(("localhost", 1200)) 
print("RPC Server started and listening on port 1200...") 
 
while True: 
    data, address = server.recvfrom(4096) 
    msg = data.decode().split() 
    if not msg: 
        continue 
    if msg[0].lower() == "q": 
        print("Shutting down server...") 
        break 
     
    method = msg[0].lower() 
    try: 
        num = int(msg[1]) 
        if method == "square": 
            result = str(num * num) 
        elif method == "squareroot": 
            result = str(math.sqrt(num)) 
        elif method == "cube": 
            result = str(num * num * num) 
        elif method == "cuberoot": 
            result = str(num ** (1/3)) 
        else: 
            result = "Unknown operation" 
    except (IndexError, ValueError): 
        result = "Error: Invalid number format or missing argument." 
         
    print(f"Processed: {method} {num if 'num' in locals() else ''} -> Result: {result}")
    server.sendto(result.encode(), ("localhost", 1300))

Prac7b:To implement Java RMI to convert the digits of a given number into words.

InterConvert.java:

import java.rmi.*; 
 
public interface InterConvert extends Remote { 
    public String convertDigit(String no) throws Exception; 
} 

ServerConvert.java:

import java.rmi.*; 
import java.rmi.server.*; 
 
public class ServerConvert extends UnicastRemoteObject implements InterConvert { 
    public ServerConvert() throws Exception { 
    } 
    public String convertDigit(String no) throws Exception { 
        String str = ""; 
        for (int i = 0; i < no.length(); i++) { 
            int p = no.charAt(i); 
            if (p == 48) { str += "zero "; } 
            if (p == 49) { str += "one "; } 
            if (p == 50) { str += "two "; } 
            if (p == 51) { str += "three "; } 
            if (p == 52) { str += "four "; } 
            if (p == 53) { str += "five "; } 
            if (p == 54) { str += "six "; } 
            if (p == 55) { str += "seven "; } 
            if (p == 56) { str += "eight "; } 
            if (p == 57) { str += "nine "; } 
        } 
        return str; 
    } 
    public static void main(String args[]) throws Exception { 
        ServerConvert s1 = new ServerConvert(); 
        Naming.bind("Wrd", s1); 
        System.out.println("Object registered...."); 
    } 
} 


ClientConvert.java:

import java.io.*; 
import java.rmi.*; 
public class ClientConvert { 
    public static void main(String args[]) throws Exception { 
        InterConvert h1 = (InterConvert) Naming.lookup("Wrd"); 
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in)); 
        System.out.print("Enter a number : "); 
        String no = br.readLine(); 
        String ans = h1.convertDigit(no); 
        System.out.println("The word representation of the entered digit is : " + ans); 
    } 
}

Prac7B:To implement a client–server application in Python that converts the digits of a given number into their word representations. 

Server.py:

import socket 
m = ['ZERO', 'ONE', 'TWO', 'THREE', 'FOUR', 'FIVE', 'SIX', 'SEVEN', 'EIGHT', 'NINE'] 
s = socket.socket() 
s.bind(('localhost', 5000)) 
s.listen(1) 
print("Aaditya server has started!!......") 
while True:
    conn, _ = s.accept()
    data = conn.recv(1024).decode()
    if data:
        ans = " ".join(m[int(c)] for c in data if c.isdigit())
        conn.send(ans.encode())
    conn.close()
